---
title: "List Call Agents"
api: "GET /v1/api/agents"
description: "Retrieve a list of all call agents with all their details."
---

Returns a plain array of every agent in the workspace, newest first, including agents whose call flow is archived. There is no pagination. Each item also carries a summary of the call flow that uses it (`flowName`, `pathway`, `flowIsLive` and so on), so filter on `flowIsLive` to find agents that can take calls.

### Headers  

<ParamField header="x-workspace-id" type="string" required>
  Workspace identifier for the API.
</ParamField>

### Response Fields

<ResponseField name="ttsModel" type="object">
  Contains the text-to-speech model details.
</ResponseField>

<ResponseField name="ttsModel.language" type="string">
  The language of the voice model (e.g., `en`).
</ResponseField>

<ResponseField name="ttsModel.provider" type="string">
  The TTS provider name (e.g., `openai`, `cartesia`).
</ResponseField>

<ResponseField name="ttsModel.voiceId" type="string">
  The unique identifier for the voice model.
</ResponseField>

<ResponseField name="ttsModel.name" type="string">
  The name of the voice model.
</ResponseField>

<ResponseField name="_id" type="string">
  Unique identifier for the agent.
</ResponseField>

<ResponseField name="workspaceId" type="string">
  Unique identifier for the workspace associated with the agent.
</ResponseField>

<ResponseField name="userId" type="string">
  Unique identifier for the user associated with the agent.
</ResponseField>

<ResponseField name="name" type="string">
  Name of the agent.
</ResponseField>

<ResponseField name="llmModel" type="string">
  The LLM (language model) used by the agent (e.g., `gpt-4o-mini`).
</ResponseField>

<ResponseField name="type" type="string">
  Type of agent (e.g., `voice`).
</ResponseField>

<ResponseField name="settings" type="object">
  The settings for the agent, including `speed`, `volume`, and `temperature`.
</ResponseField>

<ResponseField name="settings.speed" type="number">
  Speaking speed. How it is applied depends on the voice provider; `0` means the provider's normal speed.
</ResponseField>

<ResponseField name="settings.volume" type="number">
  Volume of the voice.
</ResponseField>

<ResponseField name="settings.temperature" type="number">
  Creativity of the language model, from 0 to 1.
</ResponseField>

<ResponseField name="status" type="string">
  The current status of the agent (e.g., `ready`).
</ResponseField>

<ResponseField name="phoneNumberIds" type="array">
  The agent's phone numbers as full phone number objects (not IDs). A number that belongs to another workspace is cut down to its basic fields (`_id`, `phoneNumber`, `country`, `type`, `status`, `label`, `provider`, `isExternal`).
</ResponseField>

<ResponseField name="fallbackTtsModel" type="object | null">
  Fallback voice, if one is set.
</ResponseField>

<ResponseField name="sttProvider" type="object | null">
  Speech-to-text provider (`id`, `name`, `description`), if one is set.
</ResponseField>

<ResponseField name="fallbackSttProvider" type="object | null">
  Fallback speech-to-text provider, if one is set.
</ResponseField>

<ResponseField name="fallbackLlmModel" type="string | null">
  Fallback language model, if one is set.
</ResponseField>

<ResponseField name="flowName" type="string | null">
  Name of the call flow that uses this agent, or `null` if none does.
</ResponseField>

<ResponseField name="pathway" type="string | null">
  ID of that call flow.
</ResponseField>

<ResponseField name="flowDirection" type="string | null">
  Direction of that call flow: `outbound` or `bidirectional`.
</ResponseField>

<ResponseField name="flowIsLive" type="boolean">
  Whether a number given to this agent would take calls. `false` when the agent has no call flow, or its flow is archived or inactive.
</ResponseField>

<ResponseField name="flowArchivedAt" type="string | null">
  When the call flow was archived, or `null`.
</ResponseField>

<ResponseField name="hipaaCompliance" type="boolean">
  Whether the agent's call flow has HIPAA mode on.
</ResponseField>

<ResponseField name="activeSessions" type="array">
  A list of active sessions associated with the agent.
</ResponseField>

<ResponseField name="createdAt" type="string">
  The timestamp when the agent was created.
</ResponseField>

<ResponseField name="updatedAt" type="string">
  The timestamp when the agent was last updated.
</ResponseField>
